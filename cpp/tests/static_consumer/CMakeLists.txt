cmake_minimum_required(VERSION 3.26)
project(static_tensoracle_consumer LANGUAGES C CXX)
find_package(tensoracle 2 CONFIG REQUIRED)
find_package(tensoracle 2 CONFIG REQUIRED)
get_target_property(_type tensoracle::tensoracle TYPE)
if(NOT _type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "The source build must install a static tensoracle target")
endif()
if(NOT COMMAND tensoracle_add_cubin_plugin OR NOT COMMAND tensoracle_find_cuda)
    message(FATAL_ERROR "Static installs must retain the public cubin helpers")
endif()
add_library(fixture OBJECT fixture.cpp)
target_link_libraries(fixture PRIVATE tensoracle::tensoracle TensorRT::nvinfer tensoracle_cuda_driver)
foreach(_language IN ITEMS c cpp)
    add_executable(consumer_${_language} main.${_language} $<TARGET_OBJECTS:fixture>)
    target_compile_features(consumer_${_language} PRIVATE c_std_11 cxx_std_17)
    target_link_libraries(consumer_${_language} PRIVATE tensoracle::tensoracle)
    target_compile_options(consumer_${_language} PRIVATE
        $<$<C_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Werror>)
endforeach()
