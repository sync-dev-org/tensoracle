cmake_minimum_required(VERSION 3.26)
project(miniz LANGUAGES C)
set(source "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_library(miniz STATIC "${source}/miniz.c" "${source}/miniz_zip.c"
    "${source}/miniz_tinfl.c" "${source}/miniz_tdef.c")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/miniz_export.h" "#pragma once\n#define MINIZ_EXPORT\n")
target_include_directories(miniz PRIVATE "${source}" "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(miniz PROPERTIES C_VISIBILITY_PRESET hidden)
install(TARGETS miniz ARCHIVE DESTINATION lib)
install(DIRECTORY "${source}/" DESTINATION include FILES_MATCHING PATTERN "*.h")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/miniz_export.h" DESTINATION include)
